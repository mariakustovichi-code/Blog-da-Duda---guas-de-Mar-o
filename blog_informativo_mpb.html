class BossaNovaRhythmEngine {
    constructor(bpm = 120) {
        this.bpm = bpm;
        this.syncopationPattern = [1, 0, 0, 1, 0, 1, 1, 0]; // Batida Tamborim/Violão João Gilberto
    }

    generateBossaGroove(audioCtx) {
        const sixteenthNoteTime = (60 / this.bpm) / 4;
        
        this.syncopationPattern.forEach((accent, index) => {
            if (accent) {
                const osc = audioCtx.createOscillator();
                const gain = audioCtx.createGain();
                osc.frequency.value = 220.0; // Frequência Lá 2 (A3)
                gain.gain.setValueAtTime(0.8, audioCtx.currentTime + index * sixteenthNoteTime);
                osc.connect(gain);
                gain.connect(audioCtx.destination);
            }
        });
        return "Ritmo Batida João Gilberto Sintetizado com Sucesso!";
    }
}

function analyzeMPBSubstitutions(chord) {
    // Adiciona extensões de 9ª e 13ª típicas da Bossa Nova
    if (chord.includes("7")) {
        return `${chord} -> Adicionado (9, 13) para Sofisticação Jobim`;
    }
    return chord;
}
